import { StyleSheet, Text, View, ImageBackground, Image } from "react-native";

import RecipeListIcon from "../../../assets/icons/recipeList_iconActive.svg";
import PantryIcon from "../../../assets/icons/pantry_iconActive.svg";
import GroceryListIcon from "../../../assets/icons/groceryList_iconActive.svg";
import ExpensesIcon from "../../../assets/icons/expenses_iconActive.svg";

import { LinearGradient } from "expo-linear-gradient";
import PrimaryButton from "../../components/PrimaryButton";
import Carousel from "../../components/Carousel";

/**
 * Start screen of the ListOChef application.
 *
 * Displays the app logo, slogan, navigation buttons for login and
 * registration, and a carousel presenting the app's main features.
 *
 * @module Start
 */

/**
 * Renders the main welcome screen of the app.
 *
 * This screen acts as the entry point for users and allows them to:
 * - navigate to the login screen
 * - navigate to the registration screen
 * - preview the application's main features through a carousel
 *
 * @function Start
 * @param {Object} props - Props provided by the navigator.
 * @returns {JSX.Element} The rendered start screen.
 */
const Start = (props) => {
  /**
   * Items displayed inside the feature carousel.
   *
   * Each item contains a title, short description, and icon representing
   * one of the main features of the ListOChef app.
   *
   * @type {Array<Object>}
   */
  const carouselItems = [
    {
      title: "Recipes",
      description: "Discover new recipes or create your own in seconds",
      icon: RecipeListIcon,
    },
    {
      title: "Pantry Control",
      description: "See all the ingredients you have at home at a glance",
      icon: PantryIcon,
    },
    {
      title: "Shopping List",
      description: "Add the ingredients you need and never forget a thing",
      icon: GroceryListIcon,
    },
    {
      title: "Expenses Tracking",
      description: "Track your spending by snapping a photo of your receipt",
      icon: ExpensesIcon,
    },
  ];

  /**
   * Main UI layout of the Start screen.
   *
   * Structure:
   * - Background image with gradient overlay
   * - App logo and title
   * - Slogan section
   * - Navigation buttons (Login / Register)
   * - Feature carousel
   */
  return (
    <ImageBackground source={require("../../../assets/fondoApp.png")} style={styles.background} resizeMode="cover">
      {/* Gradient overlay for background */}
      <LinearGradient colors={["rgba(241, 255, 235, 0.39)", "rgba(49, 96, 29, 0.71)"]} start={{ x: 0.5, y: 0 }} end={{ x: 0.5, y: 1 }} style={styles.overlay}>
        <View style={styles.layout}>
          {/* App logo */}
          <Image source={require("../../../assets/icons/ListoChefLogo.png")} style={styles.image} />

          {/* App title */}
          <Text style={styles.appTitle}>ListOChef</Text>

          {/* Slogan section */}
          <View style={styles.slogan}>
            <Text style={styles.textSlogan}>Plan your shopping.</Text>
            <Text style={styles.textSlogan}>Track your spending.</Text>
            <Text style={styles.textSlogan}>Cook like a chef.</Text>
          </View>

          {/* Navigation buttons */}
          <PrimaryButton buttonText={"Login"} onPress={() => props.navigation.navigate("Login")} />

          <PrimaryButton buttonText={"Register"} onPress={() => props.navigation.navigate("Register")} />

          {/* Feature carousel showcasing app functionalities */}
          <Carousel
            data={carouselItems}
            autoScroll={true}
            autoScrollInterval={4000}
            renderItem={({ item }) => (
              <View style={styles.itemCard}>
                {/* Feature icon */}
                {item.icon && <item.icon width={40} height={40} />}

                {/* Feature title */}
                <Text style={styles.itemTitle}>{item.title}</Text>

                {/* Feature description */}
                <Text style={styles.itemDescription}>{item.description}</Text>
              </View>
            )}
          />
        </View>
      </LinearGradient>
    </ImageBackground>
  );
};

/**
 * Styles for the Start screen.
 */
const styles = StyleSheet.create({

   * Background image container covering the full screen.
   */
  background: {
    flex: 1,
    justifyContent: "center",
  },

  /**
   * Gradient overlay placed above the background image.
   */
  overlay: {
    flex: 1,
  },

  /**
   * Main layout wrapper for all screen content.
   */
  layout: {
    flex: 1,
    justifyContent: "center",
    alignContent: "center",
    padding: "2%",
  },

  /**
   * Main app title text style.
   */
  appTitle: {
    fontFamily: "MontserratBold",
    fontSize: 60,
    lineHeight: 60,
    color: "#173509",
    textAlign: "center",
    textShadowColor: "rgba(0, 0, 0, 0.1)",
    textShadowOffset: { width: 0, height: 5 },
    textShadowRadius: 4,
    paddingTop: "1%",
  },

  /**
   * Wrapper for the slogan text lines.
   */
  slogan: {
    alignItems: "center",
    margin: 20,
    color: "#173509",
  },

  /**
   * Individual slogan text style.
   */
  textSlogan: {
    color: "#414141",
    fontSize: 20,
    lineHeight: 24,
    fontFamily: "MontserratBold",
  },

  /**
   * Logo image style.
   */
  image: {
    paddingTop: "5%",
    width: "50%",
    height: 180,
    resizeMode: "contain",
    alignSelf: "center",
    marginBottom: "2%",
  },

  /**
   * Card style used for each carousel item.
   */
  itemCard: {
    marginTop: 12,
    alignItems: "center",
    justifyContent: "space-evenly",
    backgroundColor: "#ecfbe5",
    borderRadius: 10,
    padding: 16,
    minHeight: 160,
    shadowColor: "#000",
    shadowOffset: { width: 0, height: 4 },
    shadowOpacity: 0.4,
    shadowRadius: 3,
    elevation: 4,
  },

  /**
   * Title text for each carousel item.
   */
  itemTitle: {
    fontSize: 20,
    fontFamily: "InterSemiBold",
    color: "#2C5818",
  },

  /**
   * Description text for each carousel item.
   */
  itemDescription: {
    fontSize: 15,
    fontFamily: "MontserratSemiBold",
    color: "black",
    textAlign: "center",
  },
});

export default Start;
